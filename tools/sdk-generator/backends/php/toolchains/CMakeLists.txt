cmake_minimum_required(VERSION 3.20)
project(admitted_php_plugin LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(protobuf_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(protobuf_BUILD_PROTOC_BINARIES OFF CACHE BOOL "" FORCE)
set(protobuf_BUILD_LIBPROTOC ON CACHE BOOL "" FORCE)
set(protobuf_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(protobuf_INSTALL OFF CACHE BOOL "" FORCE)
set(protobuf_WITH_ZLIB OFF CACHE BOOL "" FORCE)
set(ABSL_PROPAGATE_CXX_STD ON CACHE BOOL "" FORCE)
foreach(SOURCE_ROOT PROTOBUF_SOURCE GRPC_SOURCE ABSL_ROOT_DIR)
  file(TO_CMAKE_PATH "${${SOURCE_ROOT}}" ${SOURCE_ROOT})
  if(NOT IS_DIRECTORY "${${SOURCE_ROOT}}")
    message(FATAL_ERROR "Pass an admitted ${SOURCE_ROOT} source directory")
  endif()
endforeach()
add_subdirectory("${PROTOBUF_SOURCE}" protobuf)
add_executable(grpc_php_plugin "${GRPC_SOURCE}/src/compiler/php_plugin.cc" "${GRPC_SOURCE}/src/compiler/php_generator.cc" "${GRPC_SOURCE}/src/compiler/proto_parser_helper.cc")
set_property(TARGET grpc_php_plugin PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_include_directories(grpc_php_plugin PRIVATE "${GRPC_SOURCE}" "${GRPC_SOURCE}/include")
target_link_libraries(grpc_php_plugin PRIVATE protobuf::libprotoc protobuf::libprotobuf)
