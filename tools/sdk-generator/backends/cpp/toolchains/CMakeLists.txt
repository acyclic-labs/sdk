cmake_minimum_required(VERSION 3.20)
project(admitted_cpp_plugin LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
foreach(SOURCE_ROOT PROTOBUF_SOURCE PROTOBUF_BUILD GRPC_SOURCE ABSL_ROOT_DIR)
  file(TO_CMAKE_PATH "${${SOURCE_ROOT}}" ${SOURCE_ROOT})
  if(NOT IS_DIRECTORY "${${SOURCE_ROOT}}")
    message(FATAL_ERROR "Pass an admitted ${SOURCE_ROOT} directory")
  endif()
endforeach()
add_executable(grpc_cpp_plugin "${GRPC_SOURCE}/src/compiler/cpp_plugin.cc" "${GRPC_SOURCE}/src/compiler/cpp_generator.cc" "${GRPC_SOURCE}/src/compiler/proto_parser_helper.cc")
set_property(TARGET grpc_cpp_plugin PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_include_directories(grpc_cpp_plugin PRIVATE "${GRPC_SOURCE}" "${GRPC_SOURCE}/include" "${PROTOBUF_SOURCE}/src" "${ABSL_ROOT_DIR}")
file(GLOB_RECURSE SUPPORT_LIBRARIES "${PROTOBUF_BUILD}/third_party/*.lib")
target_link_libraries(grpc_cpp_plugin PRIVATE "${PROTOBUF_BUILD}/libprotoc.lib" "${PROTOBUF_BUILD}/libprotobuf.lib" ${SUPPORT_LIBRARIES})
